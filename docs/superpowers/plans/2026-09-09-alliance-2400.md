> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new OFIS section that prints one worker's удостоверение (2-page) and two справки onto the firm's own uploaded blanks, then saves three separate PDFs named after the worker.

**Architecture:** New self-contained modules (`alliance_fields`, `alliance_renderer`, `alliance_service`, `alliance_controller`, `alliance_view`) that reuse the proven building blocks already in the codebase: the ТРУД-8 field-catalogue + arranger (the office uploads its own blank and places fields on it, since blank layouts are unknown), the АЛЬПИНИСТ photo pipeline (`prepare_portrait` 3×4 + white ground) and signature ink (`ink_only`) and `plus_three_years` date, the shared `PassportReview` read-then-check panel, and per-blank layout persistence (`blank_layout`). No existing module is modified except `main_window.py` (one nav row).

**Tech Stack:** Python 3.12, PySide6, PyMuPDF (fitz), Pillow, the existing OFIS `src/` packages.

```python
from __future__ import annotations
from dataclasses import dataclass, field
from datetime import date
from pathlib import Path

from src.pdf.alliance_fields import CATALOGUE
from src.pdf.alpinist_renderer import plus_three_years

def _dots(d: date | None) -> str:
    return d.strftime("%d.%m.%Y") if d else ""

@dataclass
class AllianceData:
    surname: str = ""
    name: str = ""
    patronymic: str = ""
    gender: str = ""
    ud_number: str = ""
    dolzhnost: str = ""
    start_date: date | None = None
    photo_png: bytes | None = None
    sign_png: bytes | None = None
    layout: dict | None = None

    def fio(self) -> str:
        return " ".join(x for x in (self.surname, self.name,
                                    self.patronymic) if x).strip()

def end_date(start: date | None) -> date | None:
    return plus_three_years(start)

def values(data: AllianceData) -> dict[str, str]:
    fio = data.fio()
    return {
        "fio": fio, "fio_upper": fio.upper(),
        "surname": data.surname or "", "name": data.name or "",
        "patronymic": data.patronymic or "", "gender": data.gender or "",
        "ud_number": data.ud_number or "", "dolzhnost": data.dolzhnost or "",
        "start_date": _dots(data.start_date),
        "end_date": _dots(plus_three_years(data.start_date)),
    }

def output_name(data: AllianceData) -> str:
    stem = f"{data.surname}_{data.name}".upper().strip("_") or "ALLIANCE"
    safe = "".join(c for c in stem if c.isalnum() or c in "_-")
    return f"{safe or 'ALLIANCE'}.pdf"
```
